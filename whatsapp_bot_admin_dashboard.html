1. Servidor del Dashboard (dashboard.js)

Copia este código en tu repositorio `whatsapp-dashboard` en Render.

const express = require('express');
const http = require('http');
const { Server } = require('socket.io');
const cors = require('cors');

const app = express();
const server = http.createServer(app);
const io = new Server(server, { cors: { origin: '*' } });

app.use(cors({ origin: '*' }));
app.use(express.json());

// Memoria temporal de conversaciones
const conversations = {};

// Webhook para mensajes ENTRANTES del cliente
app.post('/api/webhook', (req, res) => {
  res.sendStatus(200);
  const body = req.body;
  
  if (body.object === 'whatsapp_business_account') {
    body.entry?.forEach(entry => {
      entry.changes?.forEach(change => {
        const value = change.value;
        if (value && value.messages) {
          value.messages.forEach(msg => {
            const phone = msg.from;
            if (!conversations[phone]) conversations[phone] = { phone, messages: [] };
            
            const isImage = msg.type === 'image';
            const imageUrl = isImage ? (msg.image?.link || msg.image?.url) : null;
            
            const newMsg = {
              id: msg.id || 'msg_' + Date.now(),
              from: phone,
              fromMe: false, // Mensaje del cliente (izquierda)
              type: isImage ? 'image' : 'text',
              text: msg.text?.body || msg.caption || '',
              mediaUrl: imageUrl,
              timestamp: msg.timestamp || Math.floor(Date.now() / 1000)
            };

            conversations[phone].messages.push(newMsg);
            io.emit('new_message', { phone, message: newMsg });
          });
        }
      });
    });
  }
});

// Endpoint para respuestas SALIENTES del Bot (derecha, verdes)
app.post('/api/bot-reply', (req, res) => {
  const { phone, text, mediaUrl } = req.body;
  if (!phone) return res.status(400).json({ error: 'Phone required' });

  const cleanPhone = String(phone).replace(/\D/g, '');
  if (!conversations[cleanPhone]) conversations[cleanPhone] = { phone: cleanPhone, messages: [] };

  const botMsg = {
    id: 'bot_' + Date.now(),
    from: cleanPhone,
    fromMe: true, // Mensaje del bot (derecha, verde)
    type: mediaUrl ? 'image' : 'text',
    text: text || '',
    mediaUrl: mediaUrl || null,
    timestamp: Math.floor(Date.now() / 1000)
  };

  conversations[cleanPhone].messages.push(botMsg);
  io.emit('new_message', { phone: cleanPhone, message: botMsg });
  res.json({ success: true });
});

// Obtener todas las conversaciones
app.get('/api/conversations', (req, res) => {
  res.json(Object.values(conversations));
});

const PORT = process.env.PORT || 10000;
server.listen(PORT, () => console.log(`Dashboard backend running on port ${PORT}`));

2. Código para el Bot Principal (`src/dashboardForwarder.js`)

Pega esto en la carpeta de tu bot (`BotMensajes`) para notificar al dashboard de forma no bloqueante.

const DASHBOARD_URL = 'https://whatsapp-dashboard-z9jm.onrender.com';

// Forwarding helper "fire-and-forget"
export function forwardToDashboard(reqBody) {
  fetch(`${DASHBOARD_URL}/api/webhook`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(reqBody)
  }).catch(err => console.error('Dashboard sync error:', err.message));
}

// Notificar respuestas salientes del bot (texto e imágenes)
export function notifyBotReply(phone, text, mediaUrl = null) {
  fetch(`${DASHBOARD_URL}/api/bot-reply`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      phone: String(phone).replace(/\D/g, ''),
      text: text || '',
      mediaUrl: mediaUrl || null
    })
  }).catch(err => console.error('Bot reply sync error:', err.message));
}
